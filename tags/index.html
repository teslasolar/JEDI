// Initialize
const tp = new TagProvider();
await tp.init();

// Query (SELECT)
tp.query("SELECT * FROM tags WHERE key='ring'");

// Execute (INSERT/UPDATE/DELETE)
tp.run("INSERT INTO tags ...", [params]);

// Directory ops
tp.addDir('/rings/r0/', 'R0', 'survival', 'R0 Survival');

// Tag ops
tp.setTag('/rings/r0/', 'prime', '2');
tp.getTag('/rings/r0/', 'prime'); // → '2'

// Script ops
tp.addScript('/', 'hello', 'return 42;');
tp.execScript('/', 'hello'); // → {ok:true, result:42}

// MCP tool
const tool = tp.asTool();

// Seed from MANIFEST.json
await tp.seedFromManifest();
